This class describes a single memory access made by a
\code{BPatch\_instruction}. It contains the information that allows the
effective address and the number of bytes transferred to be determined.
An instruction may make more than one access; \code{which} selects
among them.

\begin{apient}
  BPatch_addrSpec_NP getStartAddr_NP(int which = 0) const;
\end{apient}

\apidesc{
  Return an address specification that allows the effective address
  of a memory reference to be computed. For example,
  on the x86 platform a memory access instruction operand may contain
  a base register, an index register, a scaling
  value, and a constant base. The \code{BPatch\_addrSpec\_NP}
  describes each of these values.
}

\begin{apient}
  BPatch_countSpec_NP getByteCount_NP(int which = 0) const;
\end{apient}

\apidesc{
  Return a specification that describes the number of bytes
  transferred by the memory access.
}
